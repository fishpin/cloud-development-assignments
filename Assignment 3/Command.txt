USERS TABLE SETUP
==========================================================================================

---Create Table---------------------------------------------------------------------------
aws dynamodb create-table --table-name users --attribute-definitions AttributeName=username,AttributeType=S --key-schema AttributeName=username,KeyType=HASH --provisioned-throughput ReadCapacityUnits=5,WriteCapacityUnits=5 --endpoint-url http://localhost:4566 --region us-east-1
------------------------------------------------------------------------------------------

---Generate Password Hash (bcrypt, 12 salt rounds - run inside coupons_get_token folder)---
node -e "console.log(require('bcryptjs').hashSync('password123', 12))"

hash = $2b$12$yDLASrQVOHYwCRY1VCtKSOtsHXLpU8/b5ExlJliRsUD/2H0tC97a.
------------------------------------------------------------------------------------------

---Add User-------------------------------------------------------------------------------
aws dynamodb put-item --table-name users --% --item "{\"username\":{\"S\":\"admin\"},\"password\":{\"S\":\"$2b$12$yDLASrQVOHYwCRY1VCtKSOtsHXLpU8/b5ExlJliRsUD/2H0tC97a.\"}}" --endpoint-url http://localhost:4566 --region us-east-1
------------------------------------------------------------------------------------------

---Read User------------------------------------------------------------------------------
aws dynamodb get-item --table-name users --% --key "{\"username\":{\"S\":\"admin\"}}" --endpoint-url http://localhost:4566 --region us-east-1
